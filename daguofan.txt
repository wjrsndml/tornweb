  function gmFetch(url, json = true) {
    return new Promise((resolve, reject) => {
      if (typeof GM_xmlhttpRequest !== "function") {
        reject(new Error("GM_xmlhttpRequest is not available (missing @grant?)"));
        return;
      }
      GM_xmlhttpRequest({
        method: "GET",
        url,
        timeout: 30000,
        // 共享视图需要游客 Cookie；保留油猴默认的 Cookie/重定向行为。
        headers: { "Cache-Control": "no-cache", Accept: json ? "application/json" : "text/html" },
        onload: (resp) => {
          try {
            const status = Number(resp?.status || 0);
            if (!(status >= 200 && status < 300)) {
              reject(new Error(`gm http ${status}`));
              return;
            }
            resolve(json ? JSON.parse(String(resp?.responseText ?? "")) : String(resp?.responseText ?? ""));
          } catch (e) {
            reject(e);
          }
        },
        onerror: () => reject(new Error("gm request error")),
        ontimeout: () => reject(new Error("gm request timeout")),
        onabort: () => reject(new Error("gm request aborted")),
      });
    });
  }

  function buildFeishuTables(snapshot, records) {
    const fields = parseFeishuObject(snapshot.fieldMap, "字段");
    const cascadeFields = snapshot.viewShareEntity?.cascadeProxyFieldMap || {};
    const fieldIds = {};
    const options = {};
    for (const [id, field] of Object.entries(fields)) {
      fieldIds[String(field.name).trim()] = id;
      options[id] = new Map([
        ...(field.property?.options || []),
        ...(cascadeFields[id]?.property?.options || []),
      ].map((option) => [option.id, option.name]));
    }
    for (const name of ["OC名称", "等级", "岗位名称-简称", "成功率下限", "成功率上限", "系数", "帮派ID"]) {
      if (!fieldIds[name]) throw new Error(`飞书表缺少列：${name}`);
    }

    const cell = (record, name) => record[fieldIds[name]]?.value;
    const optionText = (name, value) => {
      const text = options[fieldIds[name]].get(value) ?? value;
      if (typeof text !== "string" || /^opt/.test(text)) throw new Error(`无法解析飞书选项：${name}`);
      return text.trim();
    };
    const number = (record, name) => {
      const value = cell(record, name);
      return (typeof value === "number" || (typeof value === "string" && value.trim())) ? Number(value) : NaN;
    };

    const tables = Object.create(null);
    for (const [recordId, record] of Object.entries(records)) {
      const factionValues = cell(record, "帮派ID");
      // 未指定帮派的记录不参与计算，避免把其他帮派或新建空行混进来。
      if (factionValues == null || (Array.isArray(factionValues) && factionValues.length === 0)) continue;
      const factionIds = (Array.isArray(factionValues) ? factionValues : [factionValues])
        .map((value) => optionText("帮派ID", value));
      const ocName = normalizeOcName(optionText("OC名称", cell(record, "OC名称")));
      const role = normalizeRole(optionText("岗位名称-简称", cell(record, "岗位名称-简称")));
      const level = number(record, "等级");
      const min = number(record, "成功率下限");
      const max = number(record, "成功率上限");
      const coeff = number(record, "系数");
      if (!ocName || !role || !Number.isInteger(level) || level <= 0 ||
          !factionIds.every((id) => /^\d+$/.test(id)) ||
          !Number.isInteger(min) || !Number.isInteger(max) || min < 0 || max > 100 || min >= max ||
          !Number.isFinite(coeff) || coeff < 0) {
        throw new Error(`飞书系数记录无效：${recordId}`);
      }
      for (const factionId of new Set(factionIds)) {
        // 注意：不要用逻辑赋值等 ES2021+ 语法，手机 PDA 的 WebView 可能很旧（见文件头注释）
        const byOc = tables[factionId] || (tables[factionId] = Object.create(null));
        const byLevel = byOc[ocName] || (byOc[ocName] = Object.create(null));
        const byRole = byLevel[String(level)] || (byLevel[String(level)] = Object.create(null));
        // 飞书是 (下限, 上限]，Torn 成功率为整数；转换成 [min, max)。
        // 例如飞书 (64, 65] -> [65, 66)，保证 65% 和 100% 都准确命中。
        (byRole[role] || (byRole[role] = [])).push([min + 1, max + 1, coeff]);
      }
    }
    for (const table of Object.values(tables)) {
      for (const levels of Object.values(table)) {
        for (const roles of Object.values(levels)) {
          for (const ranges of Object.values(roles)) ranges.sort((a, b) => a[0] - b[0]);
        }
      }
    }
    if (!isValidFactionTables(tables)) throw new Error("飞书系数为空或存在重复/重叠区间");
    return tables;
  }

  async function downloadFeishuTables() {
    const data = await fetchFeishuData("records_and_meta", {}, true);
    const snapshot = parseFeishuObject(data.snapshot, "快照");
    let page = parseFeishuObject(snapshot.viewShareEntity, "共享视图");
    const records = Object.assign(Object.create(null), parseFeishuObject(page.recordMap, "记录"));
    const total = page.recordsNum;
    if (!Number.isInteger(total) || total <= 0) throw new Error("飞书表没有系数记录");
    let pages = 1;
    while (page.hasMore === true) {
      if (++pages > 100 || !Number.isInteger(page.offsetNum) || page.offsetNum <= 0 || !page.revToken) {
        throw new Error("飞书分页信息异常");
      }
      const previousOffset = page.offsetNum;
      const next = await fetchFeishuData("records_by_page", {
        offset: page.offset || "", offsetNum: String(page.offsetNum), limit: "1000", revToken: page.revToken,
      });
      page = parseFeishuObject(next.records, "分页记录");
      const before = Object.keys(records).length;
      Object.assign(records, parseFeishuObject(page.recordMap, "记录"));
      if (Object.keys(records).length <= before || page.recordsNum !== total ||
          (page.hasMore === true && page.offsetNum <= previousOffset)) {
        throw new Error("飞书分页未完整返回或表格正在变化，请稍后重试");
      }
    }
    if (page.hasMore !== false || Object.keys(records).length !== total) {
      throw new Error("飞书系数记录不完整");
    }
    const tables = buildFeishuTables(snapshot, records);
    // 只有所有分页和校验成功后，才覆盖内存与缓存中的上一份完整数据。
    xishuTables = tables;
    xishuUpdatedAt = Date.now();
    selectXishuFaction();
    saveXishuCache();
  }

  function initializeXishu() {
    if (xishuInitialized) return;
    xishuInitialized = true;

      const saved = localStorage.getItem(XISHU_FACTION_KEY);

    } catch (e) {
      console.warn("[daguofan][xishu] load faction failed:", e);
    }
    const cached = loadXishuCache();
    if (cached) {
      xishuTables = cached.tables;
      xishuUpdatedAt = cached.ts;
      selectXishuFaction();
    }
  }

  function ensureXishuLoaded(force = false) {
    initializeXishu();
    if (xishuLoadPromise) return xishuLoadPromise;
    if (!force && Date.now() < xishuNextRefreshAt) return Promise.resolve(false);
    clearTimeout(xishuRefreshTimer);
    xishuStatus = "xishu_loading";
    xishuError = "";
    updateXishuControls();
    xishuLoadPromise = refreshXishuTable()
      .then(() => {
        xishuStatus = "xishu_ready";
        xishuNextRefreshAt = Date.now() + XISHU_REFRESH_MS;
        return true;
      })
      .catch((e) => {
        xishuStatus = "xishu_cached";
        xishuError = e.message;
        xishuNextRefreshAt = Date.now() + XISHU_RETRY_MS;
        console.warn("[daguofan][xishu] load Feishu failed, keeping last data:", e);
        return false;
      })
      .finally(() => {
        xishuLoadPromise = null;
        updateXishuControls();
        scheduleProcess();
        xishuRefreshTimer = setTimeout(() => ensureXishuLoaded(), Math.max(0, xishuNextRefreshAt - Date.now()));
      });
    return xishuLoadPromise;
  }

    // 以 slotHeader 按钮为锚点，向上取“包含 slotBody 的父级 div”

      if (w && !seen.has(w)) {
        seen.add(w);
        wrappers.push(w);
      }
    }
    return wrappers;
  }

  function parseIntSafe(v) {
    const m = String(v ?? "").match(/-?\d+/);
    return m ? Number.parseInt(m[0], 10) : NaN;
  }

  function normalizeOcName(name) {
    // 飞书选项和 Torn 的大小写可能不同，统一成小写和单空格。
    return String(name ?? "").replace(/\u00A0/g, " ").replace(/\s+/g, " ").trim().toLowerCase();
  }

  function normalizeRole(role) {
    // 飞书使用 Picklock1，Torn 显示 Picklock #1；反复归一化也保持同一结果。
    return String(role ?? "").toLowerCase().replace(/[\s#]+/g, "");
  }

  function formatProfitValue(value) {
    // 系数表里有 20.933 这样的三位小数；显示时最多保留 3 位并去掉尾随 0
    return Number(value).toFixed(3).replace(/\.?0+$/, "");
  }

  function parseRemainingDays(text) {
    const s = String(text || "").trim();
    // "5 days 17 hours 43 minutes 14 seconds"
    const m = s.match(/(\d+)\s*days?\s+(\d+)\s*hours?\s+(\d+)\s*minutes?\s+(\d+)\s*seconds?/i);
    let parts = m ? m.slice(1).map(Number) : null;
    if (!parts) {
      // "DD:HH:MM:SS"；有时也可能只有 "HH:MM:SS"
      parts = s.split(":").map((x) => Number.parseInt(x, 10));
      if (parts.length === 3) parts.unshift(0);
      if (parts.length !== 4) return null;
    }
    if (parts.some((x) => !Number.isFinite(x))) return null;
    const totalSeconds = ((parts[0] * 24 + parts[1]) * 60 + parts[2]) * 60 + parts[3];
    return totalSeconds > 0 ? totalSeconds / 86400 : null;
  }

  function getOcRemainingDays(card, vacantCount) {
    // recruiting：按空位数估算剩余天数

    if (!clockEl) return null;
    return parseRemainingDays(clockEl.getAttribute("aria-label")) || parseRemainingDays(clockEl.textContent);
  }

  function isSlotVacant(slotWrapperEl) {
    // 空缺岗位：外层常见 waitingJoin___*，slotBody 不带 validSlot，并出现 joinContainer/joinButton
    if (!slotWrapperEl) return false;

    // 强特征：出现 Join 区域 / Join 按钮 -> 认为空缺
    if (hasClassPrefix(slotWrapperEl, "waitingJoin") ||
        slotWrapperEl.querySelector(cssm("joinContainer")) ||
        slotWrapperEl.querySelector(`button${cssm("joinButton")}`)) {
      return true;
    }

    // 强特征：出现成员信息 / 个人主页链接 -> 认为已有人
    if (slotWrapperEl.querySelector(cssm("badgeContainer")) ||
        slotWrapperEl.querySelector(`${cssm("badge")}, ${cssm("textName")}`) ||
        slotWrapperEl.querySelector('a[href*="profiles.php?XID="]')) {
      return false;
    }

    // 兜底：宁可不显示，也不要误显示在“已有人岗位”
    return false;
  }

  function getXishuCoeff(ocName, level, role, chance) {
    // 约定：r 是 [min, max, a]，按 [min, max) 命中
    const ranges = XISHU_TABLE[normalizeOcName(ocName)]?.[String(level)]?.[normalizeRole(role)];
    for (const r of ranges || []) if (chance >= r[0] && chance < r[1]) return r[2];
    return null;
  }

  function getXishuMatchResult(ocName, level, role, chance) {
    const byOc = XISHU_TABLE[normalizeOcName(ocName)];
    if (!byOc) return { a: null, reason: "missing_oc" };
    const byLevel = byOc[String(level)];
    if (!byLevel) return { a: null, reason: "missing_level" };
    const ranges = byLevel[normalizeRole(role)];
    if (!Array.isArray(ranges) || ranges.length === 0) return { a: null, reason: "missing_role" };

    const a = getXishuCoeff(ocName, level, role, chance);
    if (Number.isFinite(a)) return { a, reason: "ok" };

    // 区间已按下限排序且无重叠
    const minAll = ranges[0][0];
    const maxAll = ranges[ranges.length - 1][1];
    if (chance < minAll) return { a: null, reason: "chance_too_low", minAll, maxAll };
    if (chance >= maxAll) return { a: null, reason: "chance_too_high", minAll, maxAll };
    return { a: null, reason: "no_match", minAll, maxAll };
  }

  function applyProfitHighlights(root) {
    const scoreBadges = root.querySelectorAll(`.${PROFIT_CLASS}[${BADGE_KIND_ATTR}="profit"]`);
    let maxScore = 0;
    for (const el of scoreBadges) {
      const v = Number.parseFloat(el.getAttribute(BADGE_SCORE_ATTR) || "");
      if (Number.isFinite(v)) maxScore = Math.max(maxScore, v);
    }
    if (!(maxScore > 0)) return;

    for (const el of scoreBadges) {
      el.classList.remove(HIGH_CLASS, TOP_CLASS);
      const v = Number.parseFloat(el.getAttribute(BADGE_SCORE_ATTR) || "");
      if (!Number.isFinite(v)) continue;
      if (v === maxScore) el.classList.add(TOP_CLASS);
      else if (v >= maxScore * 0.8) el.classList.add(HIGH_CLASS);
    }
  }

  function applyCardOrderByScore(ocArea, scoredCards) {
    // 不移动 DOM（React 页面移动节点风险大），仅用 CSS order 做视觉排序
    ensureOcAreaFlex(ocArea);

    const sorted = [...scoredCards].sort((a, b) => {
      const as = Number.isFinite(a.score) ? a.score : -1;
      const bs = Number.isFinite(b.score) ? b.score : -1;
      return bs !== as ? bs - as : a.originalIndex - b.originalIndex;
    });

    // order 必须是整数：按名次赋值即可
    for (let i = 0; i < sorted.length; i++) {
      const el = sorted[i].el;
      const nextOrder = String(i);
      if (el && el.style.order !== nextOrder) el.style.order = nextOrder;
    }
  }

  function processOnce() {
    if (!isTargetPage()) {
      dlog(t("log_not_target_page"), location.pathname, location.search, location.hash);
      return;
    }

    // 只有真正进入 OC 页面才联网更新系数（自带去重与30分钟节流）：
    // @match 已放宽到整站（手机 PDA 只在整页加载时注入脚本，需要靠 SPA 跳转驻留），
    // 不能在每个 Torn 页面上都请求飞书。
    ensureXishuLoaded();

    const ocArea = findOcArea();
    if (!ocArea) {
      dwarn(t("warn_missing_oc_area"));
      return;
    }

    ensureStyle();
    ensureModeToggle();
    showXishuFactionDialog();

    // 只在 Recruiting 栏位显示
    const isRecruiting = isRecruitingTabActive();
    if (!isRecruiting) {
      // 只有在“从 recruiting -> 非 recruiting”切换时才做一次性清理，避免每帧都删 DOM
      if (lastRecruitingState !== false) {
        isMutatingByScript = true;
        try {
          removeAllBadges(ocArea);
          if (lastTopCard) lastTopCard.classList.remove(CARD_TOP_CLASS);
          lastTopCard = null;
        } finally {
          isMutatingByScript = false;
        }
      }
      lastRecruitingState = false;
      dlog(t("log_not_in_recruiting"));
      return;
    }
    lastRecruitingState = true;

    const cards = getCards(ocArea);
    dlog(t("log_found_cards"), cards.length);

    // 先做读/算，最后集中写 DOM，减少抖动
    const badgeOps = [];
    const scoredCards = [];
    const runId = ++runSeq;

    for (let cardIndex = 0; cardIndex < cards.length; cardIndex++) {
      const card = cards[cardIndex];
      const ocName = normalizeOcName(card.querySelector(`p${cssm("panelTitle")}`)?.textContent);
      const level = parseIntSafe(card.querySelector(`span${cssm("levelValue")}`)?.textContent);
      if (!Number.isFinite(level)) {
        dwarn(t("warn_parse_level"), JSON.stringify(ocName));
        continue;
      }

      // 仅 7/8 级 OC 显示收益
      if (level !== 7 && level !== 8) {
        dlog(t("log_skip_non_78"), JSON.stringify(ocName), "level =", level);
        scoredCards.push({ el: card, score: -1, originalIndex: cardIndex });
        continue;
      }

      // 统计同一个 OC 的空缺岗位数 n
      const slotWrappers = getSlotWrappers(card);
      let vacantCount = 0;
      for (const w of slotWrappers) if (isSlotVacant(w)) vacantCount += 1;
      dlog(t("log_oc_slot_stats"), JSON.stringify(ocName), "level =", level, "slots =", slotWrappers.length, "vacant =", vacantCount);
      if (vacantCount <= 0) {
        scoredCards.push({ el: card, score: -1, originalIndex: cardIndex });
        continue;
      }

      const remainingDays = getOcRemainingDays(card, vacantCount);
      let cardMaxScore = -1;

      // 只对“岗位空缺”的岗位显示收益：总收益 = a * n；日均收益 = 总收益 / 剩余天数
      for (const slotWrapper of slotWrappers) {
        if (!isSlotVacant(slotWrapper)) continue;
        const btn = slotWrapper.querySelector(`:scope > button${cssm("slotHeader")}`) ||
                    slotWrapper.querySelector(`button${cssm("slotHeader")}`);
        if (!btn) continue;

        const role = normalizeRole(btn.querySelector(`span${cssm("title")}`)?.textContent);
        const chance = parseIntSafe(btn.querySelector(`div${cssm("successChance")}`)?.textContent);
        if (!Number.isFinite(chance)) {
          dwarn(t("warn_parse_chance"), JSON.stringify(ocName), JSON.stringify(role));
          continue;
        }

        const match = getXishuMatchResult(ocName, level, role, chance);
        if (match.reason === "chance_too_low") {
          badgeOps.push({ btn, text: t("warn_chance_too_low"), kind: "warn", profitValue: NaN, runId });
          continue;
        }
        if (match.reason !== "ok" || !Number.isFinite(match.a)) {
          dlog(t("log_xishu_no_hit"), { ocName, level, role, chance, vacantCount, reason: match.reason });
          continue;
        }

        const totalProfit = match.a * vacantCount;
        let score = totalProfit;
        let text = t("badge_total", { value: formatProfitValue(totalProfit) });

        if (currentMode === MODE_DAILY) {
          // 解析不到剩余时间时，兜底仍按总收益（避免除 0/NaN 导致排序全乱）
          if (Number.isFinite(remainingDays) && remainingDays > 0) score = totalProfit / remainingDays;
          text = t("badge_daily", { value: formatProfitValue(score) });
        }

        cardMaxScore = Math.max(cardMaxScore, score);
        dlog(t("log_write_profit"), { mode: currentMode, ocName, level, role, chance, a: match.a, vacantCount, remainingDays, totalProfit, score, text });
        badgeOps.push({ btn, text, kind: "profit", profitValue: score, runId });
      }

      scoredCards.push({ el: card, score: cardMaxScore, originalIndex: cardIndex });
    }

    // ====== 集中写 DOM（并在此阶段抑制 observer） ======
    isMutatingByScript = true;
    try {
      // 更新/插入 badge（尽量复用既有节点，避免 remove+insert）
      for (const op of badgeOps) upsertBadge(op.btn, op);
      // 清理本次没更新到的旧 badge（例如某岗位不再空缺/卡片被替换）
      cleanupStaleBadges(ocArea, runId);
      // 按“分数”做视觉排序（不移动节点）
      applyCardOrderByScore(ocArea, scoredCards);
      // 让高收益岗位更显眼（按当前页面最大分数做阈值）
      applyProfitHighlights(ocArea);

      // 高亮最高分卡片（只处理上一张和新一张，减少 class 抖动）
      const top = scoredCards.filter((x) => Number.isFinite(x.score) && x.score > 0)
        .sort((a, b) => b.score - a.score)[0]?.el || null;
      if (lastTopCard && lastTopCard !== top) lastTopCard.classList.remove(CARD_TOP_CLASS);
      if (top) top.classList.add(CARD_TOP_CLASS);
      lastTopCard = top;
    } finally {
      isMutatingByScript = false;
    }
  }

  function scheduleProcess() {
    if (scheduled) return;
    scheduled = true;
    requestAnimationFrame(() => {
      scheduled = false;
      try {
        processOnce();
      } catch (e) {
        // 油猴脚本里直接输出，便于定位页面结构变化
        console.error("[daguofan][oc-profit]", t("err_process_failed"), e);
      }
    });
  }

  function setupObserver() {
    if (observer) observer.disconnect();
    observer = new MutationObserver(() => {
      if (!isMutatingByScript) scheduleProcess();
    });
    // ocArea 可能延迟出现（SPA 跳转/异步渲染）：先观察根节点，DOM 变化会反复触发 processOnce
    observer.observe(findOcArea() || getCrimesRoot(), { childList: true, subtree: true });
  }

  function boot() {
    currentMode = loadMode();
    // 初次处理 + 监听后续变化
    scheduleProcess();
    setupObserver();
  }

  // Torn 是 SPA：hash 变化时重跑
  function bootEntry() {
    // 只恢复本地状态并挂监听；联网更新系数等进入 OC 页面后再做（见 processOnce）。
    initializeXishu();
    boot();
  }

  window.addEventListener("hashchange", () => bootEntry(), { passive: true });
  bootEntry();
})();
